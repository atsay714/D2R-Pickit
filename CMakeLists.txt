cmake_minimum_required(VERSION 3.28)
project(AutoPickupAnything VERSION 0.6.1 LANGUAGES CXX RC)
include(CTest)
set(D2RL_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/PluginSDK" CACHE PATH "D2RLoader Plugin SDK")
add_subdirectory("${D2RL_SDK_DIR}" sdk)
add_library(auto_pickup_anything SHARED src/plugin.cpp src/plugin.rc)
target_link_libraries(auto_pickup_anything PRIVATE D2RLPlugin::D2RLPlugin)
target_compile_features(auto_pickup_anything PRIVATE cxx_std_20)
target_compile_definitions(auto_pickup_anything PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(auto_pickup_anything PRIVATE /W4 /permissive- /Zc:__cplusplus)
target_link_options(auto_pickup_anything PRIVATE /Brepro)
set_target_properties(auto_pickup_anything PROPERTIES OUTPUT_NAME "d2rl-auto-pickup-anything" PREFIX "")
d2rlplugin_embed_config(TARGET auto_pickup_anything FILE "${CMAKE_CURRENT_SOURCE_DIR}/config/auto-pickup-anything.toml")
if(BUILD_TESTING)
    add_executable(pickup_policy_tests tests/policy_tests.cpp)
    target_compile_features(pickup_policy_tests PRIVATE cxx_std_20)
    target_compile_options(pickup_policy_tests PRIVATE /W4 /WX /EHsc)
    target_include_directories(pickup_policy_tests PRIVATE src)
    add_test(NAME auto-pickup-anything.policy COMMAND pickup_policy_tests "${CMAKE_CURRENT_SOURCE_DIR}/config/auto-pickup-anything.toml")
    add_executable(plugin_abi_tests tests/plugin_abi_tests.cpp)
    target_link_libraries(plugin_abi_tests PRIVATE D2RLPlugin::D2RLPlugin)
    target_compile_features(plugin_abi_tests PRIVATE cxx_std_20)
    target_compile_definitions(plugin_abi_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(plugin_abi_tests PRIVATE /W4 /WX /EHsc)
    target_include_directories(plugin_abi_tests PRIVATE src)
    add_dependencies(plugin_abi_tests auto_pickup_anything)
    add_test(NAME auto-pickup-anything.abi COMMAND plugin_abi_tests "$<TARGET_FILE:auto_pickup_anything>")
    add_executable(activation_tests tests/activation_tests.cpp)
    target_link_libraries(activation_tests PRIVATE D2RLPlugin::D2RLPlugin)
    target_compile_features(activation_tests PRIVATE cxx_std_20)
    target_compile_definitions(activation_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_compile_options(activation_tests PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    target_include_directories(activation_tests PRIVATE src)
    add_test(NAME auto-pickup-anything.activation COMMAND activation_tests "${CMAKE_CURRENT_SOURCE_DIR}/config/auto-pickup-anything.toml")
    add_executable(runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(runtime_tests PRIVATE D2RLPlugin::D2RLPlugin)
    target_compile_features(runtime_tests PRIVATE cxx_std_20)
    target_compile_definitions(runtime_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_compile_options(runtime_tests PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    target_include_directories(runtime_tests PRIVATE src)
    add_test(NAME auto-pickup-anything.runtime COMMAND runtime_tests)
endif()
